def topbar(title, right=""):
    return f"""
  <div style="display: flex; align-items: center; justify-content: space-between; height: 56px; padding: 0 16px; margin-top: 44px;">
    <div style="display: flex; align-items: center; gap: 8px;">{icon("back")}<div style="font-size: 18px; font-weight: 600;">{title}</div></div>

def tabbar(active):
    tabs = [("home","피드"),("bookmark","찜"),("sliders","설정"),("user","내 프로필")]
    cells = ""
    for key, label in tabs:
        on = key == active
        color = "#2d5be3" if on else "#8a877f"
        cells += f'<div style="display: flex; flex-direction: column; align-items: center; gap: 4px; flex-grow: 1; padding-top: 10px;">{icon(key, 22, color)}<div style="font-size: 11px; font-weight: {600 if on else 500}; color: {color};">{label}</div></div>'
    return f"""
  <div style="display: flex; border-top: 1px solid #e5e2db; background: #ffffff; height: 80px; margin-top: auto;">{cells}</div>"""

def section(label):
    return f'<div style="font-size: 12px; font-weight: 600; color: #8a877f; letter-spacing: 0.04em; padding: 20px 20px 8px;">{label}</div>'

def card(inner, pad="16px", gap="12px"):
    return f'<div style="background: #ffffff; border: 1px solid #e5e2db; border-radius: 14px; padding: {pad}; display: flex; flex-direction: column; gap: {gap}; margin: 0 16px;">{inner}</div>'

def row(label, value, sub=None, last=False, toggle=None):
    border = "" if last else "border-bottom: 1px solid #efece6;"
    subhtml = f'<div style="font-size: 12px; color: #8a877f; margin-top: 2px;">{sub}</div>' if sub else ""
    if toggle is None:
        right = f'<div style="display: flex; align-items: center; gap: 6px; font-size: 14px; color: #55524b;"><span>{value}</span>{icon("chev", 16, "#b8b4ab")}</div>'
    else:
        right = tog(toggle)
    return f'<div style="display: flex; align-items: center; justify-content: space-between; min-height: 52px; padding: 10px 0; {border}"><div><div style="font-size: 15px; font-weight: 500;">{label}</div>{subhtml}</div>{right}</div>'

def tog(on):
    bg = "#2d5be3" if on else "#d6d2c9"
    left = "22px" if on else "2px"
    return f'<div style="width: 44px; height: 26px; border-radius: 13px; background: {bg}; position: relative; flex-shrink: 0;"><div style="position: absolute; top: 2px; left: {left}; width: 22px; height: 22px; border-radius: 11px; background: #ffffff;"></div></div>'

def chip(label, on=False):
